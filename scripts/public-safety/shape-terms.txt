# shape-terms.txt — the generic private-metadata shapes this public repository
# screens for, one per line, as three tab-separated fields:
#
#   <class>	<kind>	<value>
#
#   class  maintainer-path | internal-address | internal-hostname
#   kind   literal | regex
#
# Shapes only. This file is published, so it can never hold a real hostname, a
# real address, a real account name, or a credential of any kind — a denylist
# of real values would publish exactly what it exists to protect. A workspace
# that must screen for named private identifiers keeps that list outside every
# public checkout and passes it to the wrapper with `--terms`.
#
# Each entry below matches a form that is true of any maintainer's machine, not
# of one particular machine.

# Absolute home directories. A path under /Users or /home names an account and
# a machine layout, and it does not work on anyone else's checkout. The
# repository-relative form or a documented placeholder is always available.
maintainer-path	regex	(^|[^A-Za-z0-9_.-])/Users/[a-z][a-z0-9._-]*/
maintainer-path	regex	(^|[^A-Za-z0-9_.-])/home/[a-z][a-z0-9._-]*/

# Private address ranges: RFC 1918 and the CGNAT range mesh VPNs allocate from.
# Matching by range rather than by address is the whole point — an address list
# would be the private topology, written down.
internal-address	regex	(^|[^0-9.])10\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}([^0-9.]|$)
internal-address	regex	(^|[^0-9.])192\.168\.[0-9]{1,3}\.[0-9]{1,3}([^0-9.]|$)
internal-address	regex	(^|[^0-9.])172\.(1[6-9]|2[0-9]|3[01])\.[0-9]{1,3}\.[0-9]{1,3}([^0-9.]|$)
internal-address	regex	(^|[^0-9.])100\.(6[4-9]|[7-9][0-9]|1[01][0-9]|12[0-7])\.[0-9]{1,3}\.[0-9]{1,3}([^0-9.]|$)

# Hostname suffixes that only resolve inside a network. A name ending in one of
# these is a machine on someone's LAN, and publishing it publishes topology.
#
# Both rows start at a token boundary, the same way every address shape above
# does. Without it the first label may begin mid-token, so a dotted filename
# such as `.env` plus a `.local` suffix is read as a host -- a file most
# repositories have, blocking an ordinary sentence about configuration. Two
# rows rather than one optional group because a middle label must not be
# allowed to supply the boundary the first label is required to have: the
# second row spells the extra labels out. The comments here name no host,
# for the same reason the test cases assemble theirs at run time.
internal-hostname	regex	(^|[^A-Za-z0-9_.-])[a-z0-9][a-z0-9-]+\.(local|internal|lan|intranet)([^A-Za-z0-9.-]|$)
internal-hostname	regex	(^|[^A-Za-z0-9_.-])[a-z0-9][a-z0-9-]+(\.[a-z0-9][a-z0-9-]+)+\.(local|internal|lan|intranet)([^A-Za-z0-9.-]|$)
